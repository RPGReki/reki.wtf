  assign pages = site.pages | sort: "url"
%}{% capture now %}{{ 'now'| date: '%s' | plus: 10800 }}{% endcapture 
%}{%
  for page in pages
%}{% capture page_time %}{{ page.date | date: '%s' }}{% endcapture 
%}{%
  if page.path contains 'assets' or 
     page.sitemap == false or 
     page_time > now or
     page.published == 'private' or 
     page.url contains 'robots.txt' or 
     page.url contains '/tags' or
     page.url contains '/.dev' or
     page.url contains '/.well-known' or
     page.url contains '.ssml' or
     page.url contains '.xml' or
     page.url contains '_schedule'
%}{% 
  continue
%}{% 
  endif
%}{{ page.url | absolute_url }}
{% endfor %}
